cmake_minimum_required(VERSION 3.25)
project(AudioGraphRealtimeExample VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The first realtime package supports Windows x64 only")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The realtime example requires a 64-bit toolchain")
endif()
if(MSVC)
    string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" RT_TARGET_ARCH)
else()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" RT_TARGET_ARCH)
endif()
if(NOT RT_TARGET_ARCH MATCHES "^(x64|x86_64|amd64)$")
    message(FATAL_ERROR "This prototype package supports x86_64 only")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json" RT_NODES_JSON)
string(REPLACE "\r\n" "\n" RT_NODES_JSON "${RT_NODES_JSON}")
string(REPLACE "\r" "\n" RT_NODES_JSON "${RT_NODES_JSON}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json")
configure_file(cmake/node_description.h.in generated/node_description.h @ONLY NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json" CONTENT "${RT_NODES_JSON}" NEWLINE_STYLE UNIX)
add_library(ag_realtime_example_plugin SHARED src/plugin.cpp)
target_compile_features(ag_realtime_example_plugin PRIVATE cxx_std_17)
target_compile_definitions(ag_realtime_example_plugin PRIVATE AG_PLUGIN_BUILD)
target_include_directories(ag_realtime_example_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ag_realtime_example_plugin PRIVATE AudioGraph::PluginSDK)
set_target_properties(ag_realtime_example_plugin PROPERTIES PREFIX "" OUTPUT_NAME "org.audiograph.realtime_example.plugin" CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ag_realtime_example_plugin PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(ag_realtime_example_plugin PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_property(TARGET ag_realtime_example_plugin APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/manifest.json.in")
add_custom_command(TARGET ag_realtime_example_plugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DPLUGIN_FILE=$<TARGET_FILE:ag_realtime_example_plugin>"
        "-DPACKAGE_DIR=${CMAKE_CURRENT_BINARY_DIR}/package/$<CONFIG>"
        "-DNODES_FILE=${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json"
        "-DPLATFORM=${CMAKE_SYSTEM_NAME}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" VERBATIM)
