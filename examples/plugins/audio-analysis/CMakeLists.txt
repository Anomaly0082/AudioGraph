cmake_minimum_required(VERSION 3.25)
project(AudioGraphAudioAnalysis VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Audio analysis requires Windows x64 with MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" ANALYSIS_ARCH)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT ANALYSIS_ARCH MATCHES "^(x64|x86_64|amd64)$")
    message(FATAL_ERROR "Audio analysis requires x86_64")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json" ANALYSIS_NODES_JSON)
string(REPLACE "\r\n" "\n" ANALYSIS_NODES_JSON "${ANALYSIS_NODES_JSON}")
string(REPLACE "\r" "\n" ANALYSIS_NODES_JSON "${ANALYSIS_NODES_JSON}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json")
configure_file(cmake/node_description.h.in generated/node_description.h @ONLY NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json" CONTENT "${ANALYSIS_NODES_JSON}" NEWLINE_STYLE UNIX)
add_library(ag_audio_analysis_plugin SHARED src/plugin.cpp)
target_compile_features(ag_audio_analysis_plugin PRIVATE cxx_std_20)
target_compile_definitions(ag_audio_analysis_plugin PRIVATE AG_PLUGIN_BUILD)
target_include_directories(ag_audio_analysis_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ag_audio_analysis_plugin PRIVATE AudioGraph::PluginSDK)
target_compile_options(ag_audio_analysis_plugin PRIVATE /W4 /permissive- /utf-8)
set_target_properties(ag_audio_analysis_plugin PROPERTIES PREFIX "" OUTPUT_NAME "org.audiograph.analysis.plugin" CXX_EXTENSIONS OFF)
set_property(TARGET ag_audio_analysis_plugin APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/manifest.json.in")
add_custom_command(TARGET ag_audio_analysis_plugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DPLUGIN_FILE=$<TARGET_FILE:ag_audio_analysis_plugin>"
        "-DPACKAGE_DIR=${CMAKE_CURRENT_BINARY_DIR}/package/$<CONFIG>"
        "-DNODES_FILE=${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json"
        "-DPLATFORM=${CMAKE_SYSTEM_NAME}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" VERBATIM)
