cmake_minimum_required(VERSION 3.25)
project(AudioGraphRNNoiseRealtime VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This experimental package supports Windows x64 only")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "A 64-bit toolchain is required")
endif()
if(MSVC)
    string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" RT_TARGET_ARCH)
else()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" RT_TARGET_ARCH)
endif()
if(NOT RT_TARGET_ARCH MATCHES "^(x64|x86_64|amd64)$")
    message(FATAL_ERROR "This experimental package supports x86_64 only")
endif()
set(AG_RNNOISE_SOURCE_DIR "" CACHE PATH "Explicit path to this repository's pinned third_party/rnnoise")
if(NOT AG_RNNOISE_SOURCE_DIR OR NOT IS_ABSOLUTE "${AG_RNNOISE_SOURCE_DIR}")
    message(FATAL_ERROR "Set AG_RNNOISE_SOURCE_DIR to the absolute repository third_party/rnnoise path")
endif()
file(REAL_PATH "${AG_RNNOISE_SOURCE_DIR}" RNNOISE_SOURCE)
foreach(required IN ITEMS CMakeLists.txt include/rnnoise.h COPYING)
    if(NOT EXISTS "${RNNOISE_SOURCE}/${required}")
        message(FATAL_ERROR "AG_RNNOISE_SOURCE_DIR is missing ${required}")
    endif()
endforeach()
add_subdirectory("${RNNOISE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/rnnoise")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json" RT_NODES_JSON)
string(REPLACE "\r\n" "\n" RT_NODES_JSON "${RT_NODES_JSON}")
string(REPLACE "\r" "\n" RT_NODES_JSON "${RT_NODES_JSON}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/nodes.json")
configure_file(cmake/node_description.h.in generated/node_description.h @ONLY NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json" CONTENT "${RT_NODES_JSON}" NEWLINE_STYLE UNIX)
add_library(ag_rnnoise_realtime_plugin SHARED src/plugin.cpp)
target_compile_features(ag_rnnoise_realtime_plugin PRIVATE cxx_std_17)
target_compile_definitions(ag_rnnoise_realtime_plugin PRIVATE AG_PLUGIN_BUILD)
target_include_directories(ag_rnnoise_realtime_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ag_rnnoise_realtime_plugin PRIVATE AudioGraph::PluginSDK rnnoise)
set_target_properties(ag_rnnoise_realtime_plugin PROPERTIES PREFIX "" OUTPUT_NAME "org.audiograph.rnnoise.plugin" CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ag_rnnoise_realtime_plugin PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(ag_rnnoise_realtime_plugin PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_property(TARGET ag_rnnoise_realtime_plugin APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/manifest.json.in"
    "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" "${RNNOISE_SOURCE}/COPYING")
add_custom_command(TARGET ag_rnnoise_realtime_plugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DPLUGIN_FILE=$<TARGET_FILE:ag_rnnoise_realtime_plugin>"
        "-DPACKAGE_DIR=${CMAKE_CURRENT_BINARY_DIR}/package/$<CONFIG>"
        "-DNODES_FILE=${CMAKE_CURRENT_BINARY_DIR}/generated/nodes.json"
        "-DCOPYING_FILE=${RNNOISE_SOURCE}/COPYING" "-DNOTICE_FILE=${CMAKE_CURRENT_SOURCE_DIR}/NOTICE"
        "-DPLATFORM=${CMAKE_SYSTEM_NAME}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/package.cmake" VERBATIM)
