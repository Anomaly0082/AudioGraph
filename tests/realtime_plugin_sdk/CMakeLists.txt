cmake_minimum_required(VERSION 3.25)
project(AudioGraphRealtimeSDKHarness LANGUAGES C CXX)
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
add_executable(ag_realtime_plugin_sdk_host host.cpp abi_c11.c)
target_link_libraries(ag_realtime_plugin_sdk_host PRIVATE AudioGraph::PluginSDK)
set_target_properties(ag_realtime_plugin_sdk_host PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(ag_realtime_plugin_sdk_host PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME ag_realtime_plugin_sdk_host COMMAND ag_realtime_plugin_sdk_host "${AG_TEST_PLUGIN_PATH}")
set_tests_properties(ag_realtime_plugin_sdk_host PROPERTIES TIMEOUT 30)
if(DEFINED AG_TEST_PLUGIN_SOURCE AND EXISTS "${AG_TEST_PLUGIN_SOURCE}/src/plugin.cpp")
    add_executable(ag_realtime_gain_allocations allocations.cpp "${AG_TEST_PLUGIN_SOURCE}/src/plugin.cpp")
    target_compile_features(ag_realtime_gain_allocations PRIVATE cxx_std_17)
    target_link_libraries(ag_realtime_gain_allocations PRIVATE AudioGraph::PluginSDK)
    target_include_directories(ag_realtime_gain_allocations PRIVATE "${AG_TEST_PLUGIN_GENERATED}")
    add_test(NAME ag_realtime_gain_allocations COMMAND ag_realtime_gain_allocations)
endif()
