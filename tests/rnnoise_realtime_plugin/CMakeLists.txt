cmake_minimum_required(VERSION 3.25)
project(RNNoiseRealtimeHarness LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
set(AG_RNNOISE_SOURCE_DIR "" CACHE PATH "Prepared vendored RNNoise directory")
set(AG_TEST_PLUGIN_PATH "" CACHE FILEPATH "Plugin DLL")
if(NOT EXISTS "${AG_RNNOISE_SOURCE_DIR}/include/rnnoise.h" OR NOT EXISTS "${AG_TEST_PLUGIN_PATH}")
  message(FATAL_ERROR "Set AG_RNNOISE_SOURCE_DIR and AG_TEST_PLUGIN_PATH to existing paths")
endif()
add_subdirectory("${AG_RNNOISE_SOURCE_DIR}" rnnoise-reference)
add_executable(ag_rnnoise_realtime_plugin_host host.cpp)
target_link_libraries(ag_rnnoise_realtime_plugin_host PRIVATE AudioGraph::PluginSDK rnnoise)
target_compile_features(ag_rnnoise_realtime_plugin_host PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ag_rnnoise_realtime_plugin_host PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME ag_rnnoise_realtime_plugin_host COMMAND ag_rnnoise_realtime_plugin_host "${AG_TEST_PLUGIN_PATH}")
set_tests_properties(ag_rnnoise_realtime_plugin_host PROPERTIES TIMEOUT 180)
