cmake_minimum_required(VERSION 3.25)
project(AudioGraphAnalysisSDKHarness LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The analysis DLL harness requires Windows x64")
endif()
find_package(AudioGraphPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
add_executable(ag_audio_analysis_sdk_host host.cpp)
target_link_libraries(ag_audio_analysis_sdk_host PRIVATE AudioGraph::PluginSDK)
target_compile_definitions(ag_audio_analysis_sdk_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(ag_audio_analysis_sdk_host PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(ag_audio_analysis_sdk_host PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
enable_testing()
if(DEFINED AG_TEST_PLUGIN_PATH AND NOT AG_TEST_PLUGIN_PATH STREQUAL "")
    add_test(NAME ag_audio_analysis_sdk_host COMMAND ag_audio_analysis_sdk_host "${AG_TEST_PLUGIN_PATH}")
    set_tests_properties(ag_audio_analysis_sdk_host PROPERTIES TIMEOUT 30)
endif()
